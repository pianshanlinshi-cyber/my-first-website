    public int getNumOfShares(int culNumOfShares) {
        if (this.side.getLabel().equals("SELL")) {
            return culNumOfShares - this.quantity.intValue();
        }
        return culNumOfShares + this.quantity.intValue();
    }

    public BigDecimal getAveAcquisitionCost(int numOfShares, BigDecimal averageCost) {
        if (this.side.getLabel().equals("SELL")) {
            return averageCost;
        }
        BigDecimal totalCost = new BigDecimal(numOfShares).multiply(averageCost).add(this.getVolume());
        BigDecimal totalQuantity = new BigDecimal(numOfShares).add(this.quantity);
        return totalCost.divide(totalQuantity, 2, RoundingMode.HALF_UP);


    }

    public int getPnL(BigDecimal aveAcquisitionCost) {
        if (this.side.getLabel().equals("SELL")) {
            return +(this.price.subtract(aveAcquisitionCost)).multiply(this.quantity).intValue();
        }
        return 0;
    }

    private static int[] maxWidthsPosition(List trades) throws IOException {
        int[] maxWidths = new int[]{15, 4, 8, 5, 11, 7, 3};
        int[] widths = new int[7];

        DecimalFormat df = new DecimalFormat("#,##0.0");
        DecimalFormat df2 = new DecimalFormat("#,##0.00");
        DecimalFormat df3 = new DecimalFormat("#,##0");
        int culNumOfShares = 0;
        int pnl = 0;
        BigDecimal aveAcquisitionCost = new BigDecimal(0);
        for (Trade trade : trades) {
            widths[0] = trade.getDateTime().toString().length();
            widths[1] = trade.getSide().getLabel().length();
            widths[2] = df.format(trade.getQuantity()).length();
            widths[3] = df.format(trade.getPrice()).length();
            culNumOfShares += trade.getQuantity().intValue();
            widths[4] = String.valueOf(trade.getNumOfShares(culNumOfShares)).length();
            aveAcquisitionCost = aveAcquisitionCost.add(trade.getAveAcquisitionCost(culNumOfShares, aveAcquisitionCost));
            widths[5] = df2.format(aveAcquisitionCost).length();
            pnl += trade.getPnL(aveAcquisitionCost);
            widths[6] = df3.format(pnl).length();

            for (int i = 0; i < 7; i++) {
                if (widths[i] > maxWidths[i]) {
                    maxWidths[i] = widths[i];
                }
            }
        }
        return maxWidths;
    }
